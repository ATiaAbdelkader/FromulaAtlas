   245→                    <th className="px-3 py-2">{copyFor(language, 'Sanitaire / Risks', 'الحماية / المخاطر')}</th>
   246→                    <th className="px-3 py-2">{copyFor(language, 'Notes', 'ملاحظات')}</th>
   247→                  </tr>
   248→                </thead>
   249→                <tbody>
   250→                  {result.weeks.map((w) => (
   251→                    <CalendarRow
   252→                      key={w.week}
   253→                      entry={w}
   254→                      area={parseFloat(area) || 1}
   255→                      note={customNotes[w.week] || ''}
   256→                      isEditing={editingWeek === w.week}
   257→                      onEdit={() => setEditingWeek(w.week)}
   258→                      onSave={(note) => { setCustomNotes(prev => ({ ...prev, [w.week]: note })); setEditingWeek(null); }}
   259→                    />
   260→                  ))}
   261→                </tbody>
   262→              </table>
   263→            </div>
   264→
   265→            {/* Export */}
   266→            <Button size="sm" onClick={exportPDF} className="h-11 w-full gap-2">
   267→              <FileText className="h-3.5 w-3.5" /> {copyFor(language, 'Export Complete Calendar (PDF)', 'تصدير التقويم الكامل (PDF)')}
   268→            </Button>
   269→
   270→            <div className="rounded-xl border bg-muted/20 p-3 text-xs leading-relaxed text-muted-foreground">
   271→              💡 {copyFor(language, 'Click ✏️ on any row to add custom notes. The calendar combines FAO-56 crop coefficients, INPV 2017 pest control data, NRC 2021 feed standards, and crop lifecycle phenology. Edit notes before exporting to PDF.', 'اضغط على ✏️ في أي صف لإضافة ملاحظات مخصصة. يجمع التقويم معاملات المحصول من FAO-56 وبيانات مكافحة الآفات من INPV 2017 ومعايير الأعلاف من NRC 2021 ومراحل نمو المحصول. عدّل الملاحظات قبل التصدير إلى PDF.')}
   272→            </div>
   273→          </div>
   274→        )}
   275→      </CardContent>
   276→    </Card>
   277→  );
   278→}

   284→function CalendarRow({ entry, area, note, isEditing, onEdit, onSave }: {
   285→  entry: CalendarEntry;
   286→  area: number;
   287→  note: string;
   288→  isEditing: boolean;
   289→  onEdit: () => void;
   290→  onSave: (note: string) => void;
   291→}) {
   292→  const { language } = useTranslation();
   293→  const [noteDraft, setNoteDraft] = useState(note);
   294→
   295→  const laborText = entry.labor.length > 0
   296→    ? entry.labor.map(l => `${l.task}`).join('; ')
   297→    : '—';
   298→  const fertText = entry.fertilization.length > 0
   299→    ? entry.fertilization.map(f => `N${f.n} P${f.p} K${f.k} (${f.method})`).join('; ')
   300→    : '—';
   301→  const riskText = entry.risks.length > 0
   302→    ? entry.risks.slice(0, 2).map(r => `${r.problem.name}: ${r.recommendedActives.slice(0, 2).map(a => a.name).join(', ')}`).join('; ')
   303→    : '—';
   304→  const irrM3 = entry.irrigation.etc * 10 * area / 1; // approx m³
   305→
   306→  const hasMilestone = Boolean(entry.milestone);
   307→  const hasContent = entry.labor.length > 0 || entry.fertilization.length > 0 || entry.risks.length > 0 || hasMilestone;
   308→
   309→  return (
   310→    <tr className={`border-t hover:bg-muted/20 ${hasMilestone ? 'bg-amber-50/40 dark:bg-amber-950/10' : ''} ${!hasContent ? 'opacity-50' : ''}`}>
   311→      <td className="px-2 py-1.5 font-mono font-bold text-center">{entry.week}</td>
   312→      <td className="px-2 py-1.5 text-[10px] text-muted-foreground">{entry.date?.slice(5) || entry.dayRange}</td>
   313→      <td className="px-2 py-1.5">
   314→        <span>{entry.stageEmoji}</span>{' '}
   315→        <span className="text-[10px]">{entry.stage}</span>
   316→        <span className="text-[9px] text-muted-foreground block">Kc {entry.kc.toFixed(2)}</span>
   317→        {entry.milestone && <span className="text-[9px] text-amber-700 dark:text-amber-400 block font-medium">{entry.milestone}</span>}
   318→      </td>
   319→      <td className="px-2 py-1.5 text-[10px] max-w-[180px]">
   320→        {entry.labor.length > 0 ? entry.labor.map((l, i) => (
   321→          <div key={i} className="flex items-start gap-1">
   322→            <span className={`text-[8px] mt-0.5 ${l.priority === 'critical' ? 'text-rose-600' : l.priority === 'recommended' ? 'text-amber-600' : 'text-muted-foreground'}`}>●</span>
   323→            <span>{l.task} <span className="text-muted-foreground">({l.laborDaysPerHa}d/ha)</span></span>
   324→          </div>
   325→        )) : <span className="text-muted-foreground">—</span>}
   326→      </td>
   327→      <td className="px-2 py-1.5 text-[10px] max-w-[150px]">
   328→        {entry.fertilization.length > 0 ? entry.fertilization.map((f, i) => (
   329→          <div key={i}>
   330→            <span className="font-mono">N{f.n} P{f.p} K{f.k}</span>
   331→            {f.ca ? <span className="text-muted-foreground"> Ca{f.ca}</span> : null}
   332→            <br />
   333→            <span className="text-[9px] text-muted-foreground">{f.sources.map(s => s.material).join(', ')}</span>
   334→          </div>
   335→        )) : <span className="text-muted-foreground">—</span>}
   336→      </td>
   337→      <td className="px-2 py-1.5 text-[10px] text-center">
   338→        <div className="font-mono">{entry.irrigation.etc.toFixed(1)} mm</div>
   339→        <div className="text-[9px] text-cyan-600">{irrM3.toFixed(0)} m³</div>
   340→      </td>
   341→      <td className="px-2 py-1.5 text-[10px] max-w-[150px]">
   342→        {entry.risks.length > 0 ? entry.risks.slice(0, 2).map((r, i) => (
   343→          <div key={i}>
   344→            <span className={r.type === 'disease' ? 'text-rose-600' : r.type === 'pest' ? 'text-amber-600' : 'text-lime-600'}>
   345→              {r.type === 'disease' ? '🦠' : r.type === 'pest' ? '🐛' : '🌿'}
   346→            </span>{' '}
   347→            <span className="font-medium">{r.problem.name}</span>
   348→            <br />
   349→            <span className="text-[9px] text-muted-foreground">→ {r.recommendedActives.slice(0, 2).map(a => a.name).join(', ')}</span>
   350→          </div>
   351→        )) : <span className="text-muted-foreground">—</span>}
   352→        {entry.risks.length > 2 && <span className="text-[9px] text-muted-foreground">+{entry.risks.length - 2} more</span>}
   353→      </td>
   354→      <td className="px-2 py-1.5 text-[10px]">
   355→        {isEditing ? (
   356→          <div className="flex gap-1">
   357→            <Textarea
   358→              aria-label={`${copyFor(language, 'Note for week', 'ملاحظة للأسبوع')} ${entry.week}`}
   359→              value={noteDraft}
   360→              onChange={e => setNoteDraft(e.target.value)}
   361→              className="min-h-[44px] h-11 text-xs"
   362→              placeholder={copyFor(language, 'Add note…', 'أضف ملاحظة…')}
   363→              autoFocus
   364→            />
   365→              <Button size="sm" variant="ghost" aria-label={`${copyFor(language, 'Save note for week', 'حفظ ملاحظة الأسبوع')} ${entry.week}`} onClick={() => onSave(noteDraft)} className="h-9 w-9 shrink-0 p-0"><Check className="h-4 w-4 text-emerald-600" /></Button>
   366→          </div>
   367→        ) : (
   368→          <div className="flex items-center gap-1">
   369→            <span className="text-[10px] italic text-muted-foreground truncate max-w-[80px]">{note || ''}</span>
   370→            <button type="button" aria-label={`${copyFor(language, 'Edit note for week', 'تعديل ملاحظة الأسبوع')} ${entry.week}`} onClick={() => { setNoteDraft(note); onEdit(); }} className="flex h-9 w-9 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"><Edit3 className="h-3.5 w-3.5" /></button>
   371→          </div>
   372→        )}
   373→      </td>
   374→    </tr>
   375→  );
   376→}
