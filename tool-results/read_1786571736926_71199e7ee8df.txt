    63→    63→// --------------------------------------------------------------------
    64→    64→// Zone-level calculations

    66→    66→function sprinklerZoneGpm(rows: SprinklerRow[]): number {
    67→    67→  return rows.reduce((sum, r) => {
    68→    68→    const n = getNozzle(r.code);
    69→    69→    if (!n) return sum;
    70→    70→    return sum + (n.flows[r.arc] || 0) * (r.count || 0);
    71→    71→  }, 0);
    72→    72→}
    73→    73→
    74→    74→function sprinklerZoneArea(rows: SprinklerRow[]): number {
    75→    75→  // rectangular grid approximation: area = spacing² × heads (square spacing)
    76→    76→  return rows.reduce((sum, r) => sum + Math.pow(r.spacing_ft || 0, 2) * (r.count || 0), 0);
    77→    77→}
    78→    78→
    79→    79→function dripZoneCalc(d: DripState) {
    80→    80→  const lineFt = Math.max(0, d.lineLength_ft);
    81→    81→  const spacingIn = Math.max(1, d.emitterSpacing_inch);
    82→    82→  const spacingFt = spacingIn / 12;
    83→    83→  const emitters = Math.floor(lineFt / spacingFt);
    84→    84→  const totalGph = emitters * Math.max(0, d.flowPerEmitter_gph);
    85→    85→  const totalGpm = totalGph / 60;
    86→    86→  const area = Math.max(0, d.area_ft2);
    87→    87→  const pr = precipitationRate(totalGpm, area);
    88→    88→  return { emitters, totalGph, totalGpm, pr };
    89→    89→}
    90→    90→
    91→    91→function bubblerZoneCalc(rows: BubblerRow[]) {
    92→    92→  // gallons per week (plant demand) → GPM (delivered in runMinutes per week)
    93→    93→  let totalGal = 0;
    94→    94→  rows.forEach(r => {
    95→    95→    const need = PLANT_WATER_NEEDS[r.plantType] || 0;
    96→    96→    totalGal += need * (r.count || 0);
    97→    97→  });
    98→    98→  // Use min(runMinutes) → worst case; otherwise weighted average
    99→    99→  const avgMin = rows.length
   100→   100→    ? rows.reduce((s, r) => s + Math.max(1, r.runMinutes), 0) / rows.length
   101→   101→    : 1;
   102→   102→  const gpm = avgMin > 0 ? totalGal / avgMin : 0;
   103→   103→  return { totalGal, gpm };
   104→   104→}
   105→   105→
   106→   106→function zoneGpm(z: Zone): number {
   107→   107→  if (z.type === 'sprinkler') return sprinklerZoneGpm(z.sprinklers);
   108→   108→  if (z.type === 'drip')      return dripZoneCalc(z.drip).totalGpm;
   109→   109→  return bubblerZoneCalc(z.bubblers).gpm;
   110→   110→}

   157→   157→export function IrrigationSystemDesigner() {

   159→   159→  const [pumpInput, setPumpInput] = useState({
   160→   160→    maxStationFlow_gpm: '0',
   161→   161→    staticHead_m: '15',
   162→   162→    maxDistance_m: '120',
   163→   163→    numberOfDutyPumps: '1',
   164→   164→    numberOfStandbyPumps: '1',
   165→   165→  });
   166→   166→
   167→   167→  const zoneFlows = zones.map(zoneGpm);

   174→   174→      maxDistance_m: parseFloat(pumpInput.maxDistance_m) || 0,
   175→   175→      numberOfDutyPumps: parseInt(pumpInput.numberOfDutyPumps) || 1,
   176→   176→      numberOfStandbyPumps: parseInt(pumpInput.numberOfStandbyPumps) || 0,
   177→   177→    });
   178→   178→  }, [pumpInput, maxZoneFlow]);
   179→   179→
   180→   180→  // Mutators ---------------------------------------------------------
   181→   181→  const addZone = (type: ZoneType) => {
   182→   182→    setZones(prev => [...prev, {
   183→   183→      id: uid(),
   184→   184→      name: `${type[0].toUpperCase()}${type.slice(1)} Zone ${prev.length + 1}`,
   185→   185→      type,
   186→   186→      sprinklers: type === 'sprinkler' ? [{ id: uid(), code: '8A', arc: 180, count: 4, spacing_ft: 8 }] : [],
   187→   187→      drip: type === 'drip' ? { ...DEFAULT_DRIP } : { ...DEFAULT_DRIP },
   188→   188→      bubblers: type === 'bubbler' ? [{ id: uid(), plantType: 'Small Tree', count: 2, runMinutes: 20 }] : [],
   189→   189→    }]);
   190→   190→  };

   192→   192→  const updateZone = (id: string, patch: Partial<Zone>) =>
   193→   193→    setZones(prev => prev.map(z => z.id === id ? { ...z, ...patch } : z));
   194→   194→
   195→   195→  const updateSprinkler = (zid: string, rid: string, patch: Partial<SprinklerRow>) =>
   196→   196→    setZones(prev => prev.map(z => z.id === zid ? {
   197→   197→      ...z, sprinklers: z.sprinklers.map(r => r.id === rid ? { ...r, ...patch } : r),
   198→   198→    } : z));
   199→   199→  const addSprinkler = (zid: string) =>
   200→   200→    setZones(prev => prev.map(z => z.id === zid ? {
   201→   201→      ...z, sprinklers: [...z.sprinklers, { id: uid(), code: '8A', arc: 180, count: 1, spacing_ft: 8 }],
   202→   202→    } : z));
   203→   203→  const removeSprinkler = (zid: string, rid: string) =>
   204→   204→    setZones(prev => prev.map(z => z.id === zid ? {
   205→   205→      ...z, sprinklers: z.sprinklers.filter(r => r.id !== rid),
   206→   206→    } : z));
   207→   207→
   208→   208→  const updateBubbler = (zid: string, rid: string, patch: Partial<BubblerRow>) =>
   209→   209→    setZones(prev => prev.map(z => z.id === zid ? {
   210→   210→      ...z, bubblers: z.bubblers.map(r => r.id === rid ? { ...r, ...patch } : r),
   211→   211→    } : z));
   212→   212→  const addBubbler = (zid: string) =>
   213→   213→    setZones(prev => prev.map(z => z.id === zid ? {
   214→   214→      ...z, bubblers: [...z.bubblers, { id: uid(), plantType: 'Small Tree', count: 1, runMinutes: 20 }],
   215→   215→    } : z));
   216→   216→  const removeBubbler = (zid: string, rid: string) =>
   217→   217→    setZones(prev => prev.map(z => z.id === zid ? {
   218→   218→      ...z, bubblers: z.bubblers.filter(r => r.id !== rid),
   219→   219→    } : z));
   220→   220→
   221→   221→  // PDF export -------------------------------------------------------
   222→   222→  const handleExport = () => {
   223→   223→    const w = window.open('', '_blank', 'noopener,noreferrer');
   224→   224→    if (!w) return;
   225→   225→    const zoneRows = zones.map((z, i) => {
   226→   226→      const gpm = zoneFlows[i].toFixed(2);
   227→   227→      if (z.type === 'sprinkler') {
   228→   228→        const area = sprinklerZoneArea(z.sprinklers);
   229→   229→        const pr = precipitationRate(zoneFlows[i], area).toFixed(2);
   230→   230→        const valve = recommendValve(zoneFlows[i]);
   231→   231→        const pipe = recommendPipeSize(zoneFlows[i]);
   232→   232→        return `<tr><td>${z.name}</td><td>Sprinkler</td><td>${z.sprinklers.length}</td><td>${gpm}</td><td>${pr}</td><td>${valve.model} ${valve.size}</td><td>${pipe}</td></tr>`;

   235→   235→        const d = dripZoneCalc(z.drip);
   236→   236→        return `<tr><td>${z.name}</td><td>Drip</td><td>${d.emitters}</td><td>${gpm}</td><td>${d.pr.toFixed(2)}</td><td>${recommendValve(zoneFlows[i]).model}</td><td>${recommendPipeSize(zoneFlows[i])}</td></tr>`;
   237→   237→      }
   238→   238→      const b = bubblerZoneCalc(z.bubblers);
   239→   239→      return `<tr><td>${z.name}</td><td>Bubbler</td><td>${z.bubblers.reduce((s, r) => s + r.count, 0)}</td><td>${gpm}</td><td>—</td><td>${recommendValve(zoneFlows[i]).model}</td><td>${recommendPipeSize(zoneFlows[i])}</td></tr>`;
   240→   240→    }).join('');
   241→   241→    w.document.write(`<!doctype html><html><head><title>Irrigation System Design</title>
   242→   242→<style>body{font-family:system-ui;padding:24px;color:#0f172a}h1{color:#4338ca;margin:0 0 4px}
   243→   243→.stat{display:inline-block;padding:10px 14px;border:1px solid #cbd5e1;border-radius:8px;margin:4px 4px 0 0;min-width:120px}
   244→   244→.stat .l{font-size:10px;color:#64748b;text-transform:uppercase;letter-spacing:.04em}
   245→   245→.stat .v{font-size:18px;font-weight:700;color:#4338ca}
   246→   246→table{width:100%;border-collapse:collapse;margin-top:12px}td,th{border:1px solid #cbd5e1;padding:6px 10px;font-size:12px;text-align:left}
   247→   247→h3{color:#4338ca;margin-top:18px}</style></head><body>
   248→   248→<h1>Irrigation System Design</h1>
   249→   249→<p style="font-size:11px;color:#64748b;margin:0 0 8px">Generated ${new Date().toLocaleString()}</p>
   250→   250→<div>
   251→   251→<div class="stat"><div class="l">Zones</div><div class="v">${zones.length}</div></div>
   252→   252→<div class="stat"><div class="l">Total flow</div><div class="v">${totalFlow.toFixed(2)} GPM</div></div>
   253→   253→<div class="stat"><div class="l">Max zone</div><div class="v">${maxZoneFlow.toFixed(2)} GPM</div></div>
   254→   254→<div class="stat"><div class="l">Pump power</div><div class="v">${pumpResult.pumpPower_hp.toFixed(2)} HP</div></div>
   255→   255→</div>
   256→   256→<h3>Zones</h3>
   257→   257→<table><thead><tr><th>Name</th><th>Type</th><th>Heads/Emitters</th><th>GPM</th><th>Precip. (in/h)</th><th>Valve</th><th>Pipe</th></tr></thead><tbody>${zoneRows}</tbody></table>
   258→   258→<h3>Pump sizing</h3>
   259→   259→<table>
   260→   260→<tr><td>Total head</td><td>${pumpResult.totalHead_m.toFixed(2)} m</td></tr>
   261→   261→<tr><td>Pressure</td><td>${pumpResult.pressure_psi.toFixed(1)} psi (${psiToBar(pumpResult.pressure_psi).toFixed(2)} bar)</td></tr>
   262→   262→<tr><td>Flow per pump</td><td>${pumpResult.pumpFlow_gpm.toFixed(2)} GPM</td></tr>
   263→   263→<tr><td>Power</td><td>${pumpResult.pumpPower_hp.toFixed(2)} HP (${hpToKw(pumpResult.pumpPower_hp).toFixed(2)} kW)</td></tr>
   264→   264→</table>
   265→   265→</body></html>`);
   266→   266→    w.document.close();
   267→   267→    setTimeout(() => w.print(), 300);
   268→   268→  };
   269→   269→
   270→   270→  // Render -----------------------------------------------------------
   271→   271→  return (
   272→   272→    <Card>
   273→   273→      <CardHeader className="pb-3">
   274→   274→        <CardTitle className="flex items-center gap-2 text-base">
   275→   275→          <Settings className="h-4 w-4 text-indigo-600" /> Irrigation System Designer
   276→   276→        </CardTitle>
   277→   277→        <CardDescription className="text-xs">
   278→   278→          Multi-zone sprinkler / drip / bubbler designer with valve &amp; pipe sizing, pump selection, and PDF export.
   279→   279→        </CardDescription>
   280→   280→      </CardHeader>
   281→   281→      <CardContent className="space-y-4">
   282→   282→        {/* Summary cards */}
   283→   283→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   284→   284→          {[
