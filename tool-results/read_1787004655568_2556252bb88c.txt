   221→/**
   222→ * Inline translator for tool titles and descriptions.
   223→ * French uses the curated overlay and safely falls back to English for any
   224→ * domain string not yet translated; Arabic keeps its existing source copy.
   225→ */
   226→function tr(en: string, ar: string, language: Language): string {
   227→  if (language === 'ar') return ar;
   228→  if (language === 'fr') return FRENCH_TOOL_COPY[en] ?? en;
   229→  return en;
   230→}
